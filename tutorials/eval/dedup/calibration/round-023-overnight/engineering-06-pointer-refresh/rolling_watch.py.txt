"""Persist completed evidence and immutable milestone snapshots without issuing inference."""
import hashlib
import json
import time
import sys
from fractions import Fraction
from pathlib import Path

ROOT = Path(__file__).resolve().parent
CHECKOUT = Path(json.loads((ROOT / 'SESSION_STATUS.json').read_text())['workspace'])
sys.path[:0] = [str(CHECKOUT / 'tutorials/eval/dedup'), str(CHECKOUT)]
from ledger_audit_core import strict_run
STRICT_SHA = hashlib.sha256((ROOT / 'ledger_audit_core.py').read_bytes()).hexdigest()
NAMES = ('milestone_85_80', 'target_90_85', 'best_metric', 'best_eligible')
index = {'events': [], 'runs': {}}
if (ROOT / 'rolling-index.json').exists():
    index = json.loads((ROOT / 'rolling-index.json').read_text())
pointers = json.loads((ROOT / 'pointers.json').read_text()) if (ROOT / 'pointers.json').exists() else dict.fromkeys(NAMES)
milestone_records = json.loads((ROOT / 'milestone-records.json').read_text()) if (ROOT / 'milestone-records.json').exists() else {}
(ROOT / 'snapshots').mkdir(exist_ok=True)


def read(path):
    return json.loads(path.read_text())


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, data):
    target = ROOT / name
    temporary = target.with_suffix(target.suffix + '.tmp')
    temporary.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    temporary.replace(target)


def metric_rank(metric):
    direction = Fraction(metric['correct_directions'], metric['directions'])
    pair = Fraction(metric['correct_pairs'], metric['pairs'])
    confirmed = metric['repeats'] == 3
    return (confirmed and direction >= Fraction(9, 10) and pair >= Fraction(17, 20), confirmed and direction >= Fraction(17, 20) and pair >= Fraction(4, 5), direction, pair, metric['repeats'])


def snapshot(pointer, version, status, runs, metric, promote=True):
    old = pointers[pointer]
    signature = [version, status, metric['correct_directions'], metric['correct_pairs'], len(runs), STRICT_SHA]
    if old and old['signature'] == signature:
        return old['path']
    name = f'snapshots/{time.time_ns()}-{pointer}-{version}-{status}.json'
    manifests = {run['run']: read(ROOT / version / 'runs' / run['run'] / 'run_manifest.json') for run in runs}
    related = [value for key, value in index['runs'].items() if Path(key).parts[0] == version]
    plan = read(ROOT / version / 'verification-plan.json')
    completed = {value['run'] for value in related}
    row_artifacts = {}
    for value in related:
        run_dir = ROOT / version / 'runs' / value['run']
        paths = [*[Path(path) for path in value.get('strict_artifacts', {}).values()], run_dir / 'tuning_audit-v2.csv', run_dir / 'report.json', run_dir / 'run_manifest.json']
        row_artifacts[value['run']] = {}
        for path in paths:
            if not path.exists():
                continue
            content = path.read_bytes()
            sha = hashlib.sha256(content).hexdigest()
            immutable = ROOT / 'artifact-store' / sha / path.name
            immutable.parent.mkdir(parents=True, exist_ok=True)
            if not immutable.exists():
                immutable.write_bytes(content)
            assert digest(immutable) == sha
            row_artifacts[value['run']][str(immutable)] = sha
    hard = {}
    for group in ['original36', 'extra6', 'latest18', 'controlled26', 'real_consensus6']:
        observed = [value for value in related if group in value['groups']]
        failed = any(value['groups'][group][layer]['correct_pairs'] != value['groups'][group][layer]['pairs'] or value['groups'][group][layer]['invalid_pairs'] or value['groups'][group][layer]['unresolved_pairs'] for value in observed for layer in ['main', 'coverage', 'final'])
        hard[group] = {'status': 'FAIL' if failed else 'PASS' if len(observed) == 3 else 'NOT_COMPLETED', 'completed_repeats': len(observed), 'runs': [value['run'] for value in observed]}
    development = {}
    for group in ['development99_original', 'development99_reviewed', 'development97_strict']:
        selected = [value for value in related if group in value['groups']]
        baseline = [value for key, value in index['runs'].items() if Path(key).parts[0] == 'baseline' and group in value['groups']]
        a = sum(value['groups'][group]['final']['correct_directions'] for value in selected)
        b = sum(value['groups'][group]['final']['correct_directions'] for value in baseline)
        development[group] = {'status': ('PASS' if a >= b else 'FAIL') if len(selected) == len(baseline) == 3 else 'NOT_COMPLETED', 'candidate_completed_repeats': len(selected), 'baseline_completed_repeats': len(baseline), 'candidate_correct_directions': a, 'baseline_correct_directions': b}
    code_path = ROOT / version / 'frozen-code.json'
    body = {'pointer': pointer, 'candidate': version, 'status': status, 'time': time.time(), 'metric': metric, 'run_evidence': runs, 'all_candidate_run_evidence': related, 'run_manifests': manifests, 'row_result_artifacts': row_artifacts, 'repeat_completion': {name: 'COMPLETE' if name in completed else 'NOT_COMPLETED' for names in plan['runs'].values() for name in names}, 'hard_regressions': hard, 'development_nonregression': development, 'frozen_code': read(code_path) if code_path.exists() else None, 'frozen_manifest': read(ROOT / version / 'manifest.json'), 'verification_plan': plan, 'protocol_sha256': digest(ROOT / 'frozen-evaluation-protocol.json'), 'acceptance_passed': False, 'unmet': 'Hard regressions, all-input technical validity and every development nonregression view require their own complete evidence; a metric snapshot alone is not acceptance.'}
    save(name, body)
    (ROOT / name.replace('.json', '.txt')).write_text(f"{pointer}: {version} — {status}\nFinal direction / 方向: {metric['correct_directions']}/{metric['directions']}\nFinal pair / 整对: {metric['correct_pairs']}/{metric['pairs']}\nRepeats / 重复数: {metric['repeats']}\nHard regressions / 硬回归: {json.dumps(hard, ensure_ascii=False)}\nDevelopment nonregression / 开发集不退步: {json.dumps(development, ensure_ascii=False)}\nAcceptance is not confirmed. 尚未确认验收通过。\n")
    if not promote:
        with (ROOT / 'pointer-history.jsonl').open('a') as stream:
            stream.write(json.dumps({'time': time.time(), 'event': 'retained_candidate_snapshot', 'pointer': pointer, 'candidate': version, 'status': status, 'path': name, 'promoted': False}) + '\n')
        return name
    pointers[pointer] = {'path': name, 'sha256': digest(ROOT / name), 'signature': signature, 'candidate': version, 'status': status, 'metric': metric, 'strict_audit_sha256': STRICT_SHA}
    with (ROOT / 'pointer-history.jsonl').open('a') as stream:
        stream.write(json.dumps({'time': time.time(), 'pointer': pointer, 'previous': old, 'current': pointers[pointer]}) + '\n')
    save('pointers.json', pointers)
    return name


initial_refresh = True
while not (ROOT / 'STOP_WATCH').exists():
    changed = initial_refresh
    initial_refresh = False
    for path in sorted(ROOT.glob('*/runs/*/tuning_audit-v2.json')):
        key = str(path.relative_to(ROOT))
        try:
            value = read(path)
        except (OSError, ValueError):
            continue
        if key not in index['runs'] or index['runs'][key].get('strict_audit_sha256') != STRICT_SHA or index['runs'][key]['native_report_sha256'] != value['native_report_sha256']:
            event = 're_audited_run' if key in index['runs'] else 'completed_run'
            value = strict_run(ROOT / Path(key).parts[0], value['run'], native=value)
            index['runs'][key] = value
            index['events'].append({'time': time.time(), 'event': event, 'path': key, 'sha256': digest(path), 'strict_audit_sha256': STRICT_SHA})
            changed = True
    if changed:
        if (ROOT / 'pointers.json').exists():
            pointers = read(ROOT / 'pointers.json')
        save('rolling-index.json', index)
        versions = sorted({Path(key).parts[0] for key in index['runs']})
        scores = []
        for version in versions:
            runs = [value for key, value in index['runs'].items() if Path(key).parts[0] == version and 'personal120' in value['groups']]
            if not runs:
                continue
            count = len(runs)
            metric = {'correct_directions': sum(run['groups']['personal120']['final']['correct_directions'] for run in runs), 'directions': 240 * count, 'correct_pairs': sum(run['groups']['personal120']['final']['correct_pairs'] for run in runs), 'pairs': 120 * count, 'repeats': count}
            scores.append({'candidate': version, **metric})
            if not version.startswith('candidate-'):
                continue
            for pointer, direction, pair in [('milestone_85_80', Fraction(17, 20), Fraction(4, 5)), ('target_90_85', Fraction(9, 10), Fraction(17, 20))]:
                single_key = STRICT_SHA + ':' + version + ':' + pointer + ':single'
                for run in runs:
                    single = run['groups']['personal120']['final']
                    if Fraction(single['correct_directions'], 240) >= direction and Fraction(single['correct_pairs'], 120) >= pair:
                        if single_key not in milestone_records:
                            old = pointers[pointer]
                            milestone_records[single_key] = snapshot(pointer, version, 'provisional_single_run', [run], {'correct_directions': single['correct_directions'], 'directions': 240, 'correct_pairs': single['correct_pairs'], 'pairs': 120, 'repeats': 1}, promote=not old or old['status'] != 'confirmed_three_run_mean')
                            save('milestone-records.json', milestone_records)
                if count == 3 and Fraction(metric['correct_directions'], metric['directions']) >= direction and Fraction(metric['correct_pairs'], metric['pairs']) >= pair:
                    old = pointers[pointer]
                    key = STRICT_SHA + ':' + version + ':' + pointer + ':confirmed'
                    if key not in milestone_records:
                        promote = not old or old['status'] != 'confirmed_three_run_mean' or (metric['correct_directions'], metric['correct_pairs']) > (old['metric']['correct_directions'], old['metric']['correct_pairs'])
                        milestone_records[key] = snapshot(pointer, version, 'confirmed_three_run_mean', runs, metric, promote=promote)
                        save('milestone-records.json', milestone_records)
                elif count == 3 and single_key in milestone_records:
                    key = STRICT_SHA + ':' + version + ':' + pointer + ':not_confirmed'
                    if key not in milestone_records:
                        old = pointers[pointer]
                        milestone_records[key] = snapshot(pointer, version, 'not_confirmed_three_run_mean', runs, metric, promote=bool(old and old['candidate'] == version and old['status'] != 'confirmed_three_run_mean'))
                        save('milestone-records.json', milestone_records)
        candidate_scores = [metric for metric in scores if metric['candidate'].startswith('candidate-')]
        if candidate_scores:
            selected = max(candidate_scores, key=metric_rank)
            version = selected['candidate']
            metric = {key: value for key, value in selected.items() if key != 'candidate'}
            runs = [value for key, value in index['runs'].items() if Path(key).parts[0] == version and 'personal120' in value['groups']]
            snapshot('best_metric', version, 'confirmed_three_run_mean' if metric['repeats'] == 3 else 'provisional', runs, metric)
        save('ROLLING_STATUS.json', {'updated': time.time(), 'deadline_utc': '2026-10-28T16:00:00Z', 'completed_runs': len(index['runs']), 'personal_scores': scores, 'pointers': pointers, 'best_eligible': pointers['best_eligible'], 'note': 'Eligibility requires its own complete fixed acceptance evidence.'})
    time.sleep(5)
