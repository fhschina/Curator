def percent(value):
    return f'{value * 100:.2f}%'


def table(key, en_headers, zh_headers, rows):
    tables[key] = {'headers_en': en_headers, 'headers_zh': zh_headers, 'rows': rows}
    return key

def render_table(key, language):
    value = tables[key]
    headers = value['headers_' + language]
    clean = lambda cell: str(cell[language] if isinstance(cell, dict) and language in cell else cell).replace('|', '&#124;').replace('\n', ' ').replace('{', '&#123;').replace('}', '&#125;')
    return '\n'.join(['| ' + ' | '.join(headers) + ' |', '| ' + ' | '.join(['---'] * len(headers)) + ' |', *['| ' + ' | '.join(clean(cell) for cell in row) + ' |' for row in value['rows']]])

for language in ['en', 'zh']:
    title = ('Qwen retention calibration v3 — overnight progress' if args.phase == 'progress' else 'Qwen retention calibration v3 — overnight delivery') if language == 'en' else ('Qwen 保留规则校准 v3 — 夜间进展' if args.phase == 'progress' else 'Qwen 保留规则校准 v3 — 夜间交付')

                  '<title>' + html.escape(title) + '</title><h1>' + html.escape(title) + '</h1>',
                  '<p><code>' + html.escape(state['branch'] + ' · ' + best) + '</code></p>']
    for key, en_title, zh_title, en_text, zh_text, keys in sections:
        text += f'## {en_title if language == "en" else zh_title}\n\n{en_text if language == "en" else zh_text}\n\n'
        html_parts.append('<h2>' + html.escape(en_title if language == 'en' else zh_title) + '</h2><p>' + html.escape(en_text if language == 'en' else zh_text).replace('\n', '<br>') + '</p>')
        for table_key in keys:
            value = tables[table_key]
            cell_value = lambda cell: str(cell[language] if isinstance(cell, dict) and language in cell else cell)
            html_table = '<table><thead><tr>' + ''.join('<th>' + html.escape(header) + '</th>' for header in value['headers_' + language]) + '</tr></thead><tbody>' + ''.join('<tr>' + ''.join('<td>' + html.escape(cell_value(cell)) + '</td>' for cell in row) + '</tr>' for row in value['rows']) + '</tbody></table>'
